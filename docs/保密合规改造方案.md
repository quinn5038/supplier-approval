# 保密合规改造方案

> 制定时间：2026-09-04
> 触发场景：保密专员确认 — **供应商未公开披露的财报、类似法人身份证的敏感文件不能直接发给AI处理**；公开披露信息（企查查可查到的）可发给第三方AI。
> 实施范围：参赛代码仓库 v1.1 改造

---

## 一、改造背景与依据

### 1.1 保密专员回复要点
- ❌ **禁止发给AI**：财报（未公开披露）、法人身份证（敏感证件）
- ✅ **可以发给AI**：公开披露信息，尤其企查查能查到的工商/司法/经营异常等
- ✅ **替代方案**：财报调企查查，没有上年度数据转人工不走 TextIn

### 1.2 与 161 号标准（公开部分）的差异
原《中港采购发〔2025〕161 号管理办法》要求：
- A08 上年度经审计财报 → **指标核算判定**（原：TextIn 解析供应商上传的财报）
- A02 法人身份证明 → **身份证信息与联系信息栏法人一致性**（原：身份证 OCR + 比对）

新方案（保密优先）：
- A08 财报 → **企查查财务数据**（公查到的），无数据转人工（要求供应商补交）
- A02 法人身份 → **仅校验材料是否上传**（OCR 不做），缺失→退回，上传→转人工复核（不 OCR）

## 四、改造点 3：离线材料脱敏助手

### 4.1 设计目标
> 从系统上扒下来的材料先脱敏，然后再 textin / AI 识别

### 4.2 设计原则
- **本地纯 Python 处理**——不调任何云端 OCR/AI（这就是"脱敏"的核心）
- 输出文件名加 `_desens` 后缀，与原文件区分
- textin_pipeline.py 改读 `cache_v4_desens/` 而不是 `cache_v4/`

### 4.3 模块设计

新建 `desensitize.py`（~250 行）：

```python
"""离线材料脱敏助手
本地纯 Python 处理：把从 ICCEC 系统下载的敏感材料做脱敏，
输出到 cache_v4_desens/ 给后续 textin / AI 识别使用。

按文件类型分四类处理：
- 财报 PDF：保留数字（财务指标需要），人名/签字/银行账号打码
- 身份证 JPG/PNG：人脸马赛克 + 姓名/证号/地址打码
- 营业执照：原样保留（公开信息）
- 其他（ISO/授权/声明）：原样保留
"""
import os
import shutil
from pathlib import Path

# === 入口 ===
def desensitize_dir(src_dir, dst_dir):
    """脱敏整个目录的供应商材料"""
    src = Path(src_dir)
    dst = Path(dst_dir)
    if not src.exists():
        return
    if dst.exists():
        shutil.rmtree(dst)  # 清空旧的脱敏目录
    dst.mkdir(parents=True)
    for f in src.rglob("*"):
        if f.is_file():
            rel = f.relative_to(src)
            dst_file = dst / rel
            dst_file.parent.mkdir(parents=True, exist_ok=True)
            _desensitize_one(f, dst_file)

def _desensitize_one(src, dst):
    """根据文件类型分派脱敏函数"""
    ext = src.suffix.lower()
    name = src.name.lower()
    if ext == ".pdf":
        if any(k in name for k in ["财务", "审计", "财报", "审计报告"]):
            _desensitize_financial_pdf(src, dst)
        else:
            shutil.copy2(src, dst)  # 其他 PDF 原样保留
    elif ext in (".jpg", ".jpeg", ".png"):
        if any(k in name for k in ["身份证", "id_", "证件"]):
            _desensitize_id_card_image(src, dst)
        else:
            shutil.copy2(src, dst)  # 其他图片原样保留
    else:
        shutil.copy2(src, dst)

# === 财报 PDF 脱敏 ===
def _desensitize_financial_pdf(src, dst):
    """财报 PDF 脱敏：保留数字表格，打码人名/签字/银行账号"""
    try:
        import pdfplumber
        import fitz  # PyMuPDF
        from PIL import Image, ImageDraw, ImageFilter
    except ImportError:
        shutil.copy2(src, dst)
        return  # 缺库则原样复制（不阻塞流程）

    # 简化方案——读出文本，对人名/账号字段做字符级打码
    doc = fitz.open(str(src))
    for page in doc:
        text_dict = page.get_text("dict")
        for line in text_dict.get("blocks", []):
            for span in line.get("lines", []):
                for char in span.get("spans", []):
                    text = char.get("text", "")
                    if _is_sensitive_text(text):
                        # 用白色矩形覆盖该 span
                        rect = fitz.Rect(char["bbox"])
                        page.add_redact_rect(rect, fill=(1, 1, 1))
                        # 写"***"替代
                        page.insert_text((rect.x0, rect.y1 - 2), "***",
                                        fontsize=char["size"], color=(0, 0, 0))
        page.apply_redactions()
    doc.save(str(dst))
    doc.close()

def _is_sensitive_text(text: str) -> bool:
    """判断是否敏感字段"""
    import re
    # 银行账号（19 位数字）
    if re.search(r"\d{19}", text):
        return True
    # 身份证号（18 位含 X）
    if re.search(r"\d{17}[\dXx]", text):
        return True
    # 签字标记（"签字"、"盖章"、"经办人"、"复核人"）
    if any(k in text for k in ["签字", "盖章", "经办人", "复核人", "主管"]):
        return True
    return False

# === 身份证图片脱敏 ===
def _desensitize_id_card_image(src, dst):
    """身份证图片脱敏：人脸马赛克 + 姓名/证号/地址打码"""
    try:
        from PIL import Image, ImageDraw, ImageFilter
    except ImportError:
        shutil.copy2(src, dst)
        return

    img = Image.open(src).convert("RGB")
    w, h = img.size

    # 人脸区域估算（身份证正面左上 1/4 区域，通常 60-200px, 100-300px）
    # 简化方案——对左上 1/4 区域整体马赛克
    face_box = (int(w * 0.05), int(h * 0.10), int(w * 0.35), int(h * 0.45))
    face = img.crop(face_box)
    # 缩小再放大 = 马赛克
    small = face.resize((20, 20))
    face_mosaic = small.resize(face.size, Image.NEAREST)
    img.paste(face_mosaic, face_box)

    # 姓名/证号/地址区域——下半部分文字区域打码
    # 简化方案——下半部分整片加白色遮罩
    draw = ImageDraw.Draw(img)
    draw.rectangle((int(w * 0.3), int(h * 0.45), w, h), fill=(255, 255, 255))

    img.save(dst, quality=85)


# === 命令行入口 ===
if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="离线材料脱敏助手")
    parser.add_argument("--src", default="cache_v4", help="源目录（默认 cache_v4）")
    parser.add_argument("--dst", default="cache_v4_desens", help="目标目录（默认 cache_v4_desens）")
    args = parser.parse_args()
    desensitize_dir(args.src, args.dst)
    print(f"[OK] 脱敏完成：{args.src} → {args.dst}")
```
